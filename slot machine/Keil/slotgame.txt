//slot game 老虎机相关函数
#include <reg52.h>
#include <stdlib.h>
#include "all_x.h"


int speed[][3] = {{0,5,-1},{0,7,-1},{0,9,-1},{0,11,-1},{0,12,-1},{0,13,-1}};	//在游戏开始时，每位的初始循环速度都应该不同，这里单独控制每一位的循环速度，达到丝滑效果，这个值是一直变化的,每秒减去14

	
uchar Lucky_alphabet[5] = "XXXXX";
uchar i = 0;


void slotgame()  //老虎机游戏总控
{
	uchar how_r = 0; 
	LcdShowInit("Top score:000000", 1, 0);
	
		while(now_mod == 'G')  //点击右键R开始游戏，左键L退出
		{			
			get_button(5);   //检测按键，以便于开始或者退出

			if(now_button == 'l')  //刚进入模式，单击左键退出当前模式
			{
				TR1 = 0; 				//关闭定时器
				now_mod = 'g';		//将状态调整到上一个状态
				break;
			}
			
			else if(now_button == 'r') 								//刚进入模式，单击右键开始游戏
			{
				switch(how_r)
				{
					case 0:
						LcdShowInit("Your Lucky:XXXXX", 2, 0);  //奠定一下基础，先把不变的东西写上去				
						how_r++;
					break;
					case 1:
						LcdShowInit("Your Lucky:MMMMM", 2, 0);
						TR1 = 1;    //开启定时器,游戏在中断中
						how_r++;
						break;
					case 2:
						TR1 = 0;    
						break;
				}
			}				
		}			
}



void Timer1_ISR() interrupt 3      //定时器1中断
{
	gaming();
	
	TH1 = 0xDC; ;         // 手动重装初值
  TL1 = 0x00;
	
}



void gaming()    //进行游戏
{
		for(i=0;i<5;i++)    
		{
			if(++speed[i][0] > speed[i][1])
			{
				Lucky_alphabet[i] = 'A' + wj_rand();   //每次显示前都先随机一个字母
				
				LCD_one_data(Lucky_alphabet[i], 2, i+11);   //逐位显示修改的内容

				
				speed[i][0] = 0;
				speed[i][1] += speed[i][2];
			}
		}	
}






void int_timer1()     //定时器1的初始化
{
    // 模式配置（不影响定时器0的设置）
    TMOD &= 0x0F;       // 清除T1配置位
    TMOD |= 0x10;       // 设置模式1
    
		 // 计算50ms定时初值（11.0592MHz）
    TH1 = 0xDC;         // 高位初值  65536 - 56320 = 9216  9216 * 1.085069444us = 9,999.99999590us = 10ms
    TL1 = 0x00;         // 初始值								  12 / 11.0592MHZ = 1.08506944444us↑						
    
    // 中断配置
    ET1 = 1;            // 暂时关闭定时器1中断
    TR1 = 0;            // 暂时关闭定时器1
	
	  //EA = 1;         // 开启全局中断,定时器0到是已经开启过了

}




uint wj_rand() 	//生成随机数
{
		uint random_nm;
		uchar low, high;
       
		TR1 = 0;  
	
    high = TH1;  //利用定时器的不断变换的高低八位来随机生成随机数
    low = TL1;		//定时器低八位
		srand(low<<8 | high);    //创意修改，让低位成为高位，使得每次调用时，srand相差更大
		random_nm = rand() % 3;  //生成0-2之间的随机数
 
		TR1 = 1;
  
		return random_nm;		//返回生成的随机数
}
